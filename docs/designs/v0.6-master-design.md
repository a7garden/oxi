## 0.2 Secret<T> 타입

### oxicode-ai/src/secret.rs (NEW)

```rust
//! 민감 정보(API 키 등)의 실수 노출을 방지하는 래퍼 타입.
//!
//! `Secret<T>`는 [`Debug`], [`Display`] 구현에서 값을 마스킹한다.
//! 실제 값은 [`expose()`](Secret::expose)로만 접근 가능하다.

use std::fmt;

/// API 키 등 민감 정보를 감싸는 타입.
///
/// # Examples
/// ```ignore
/// let key = Secret::new("sk-ant-1234567890abcdef".to_string());
/// println!("{key:?}");   // Secret([REDACTED])
/// println!("{key}");     // sk-an...cdef
/// let header = format!("Bearer {}", key.expose());
/// ```
#[derive(Clone, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct Secret<T> {
    inner: T,
}

impl<T> Secret<T> {
    /// 민감 값을 감싼다.
    pub fn new(value: T) -> Self {
        Self { inner: value }
    }

    /// 실제 값에 접근. HTTP 헤더 생성 등 필수 로직에서만 사용.
    pub fn expose(&self) -> &T {
        &self.inner
    }

    /// 소유권 이전으로 값 꺼내기.
    pub fn expose_owned(self) -> T {
        self.inner
    }

    /// 내부 값 변환.
    pub fn map<U>(self, f: impl FnOnce(T) -> U) -> Secret<U> {
        Secret::new(f(self.inner))
    }
}

// ── 마스킹 구현 ──────────────────────────────────────────

impl<T> fmt::Debug for Secret<T> {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.write_str("Secret([REDACTED])")
    }
}

impl fmt::Display for Secret<String> {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        let s = &self.inner;
        if s.len() > 8 {
            write!(f, "{}...{}", &s[..4], &s[s.len() - 4..])
        } else {
            f.write_str("[REDACTED]")
        }
    }
}

// ── 직렬화: 저장/전송 시 값 그대로 유지 ─────────────────

impl serde::Serialize for Secret<String> {
    fn serialize<S: serde::Serializer>(&self, s: S) -> Result<S::Ok, S::Error> {
        s.serialize_str(&self.inner)
    }
}

impl<'de> serde::Deserialize<'de> for Secret<String> {
    fn deserialize<D: serde::Deserializer<'de>>(d: D) -> Result<Self, D::Error> {
        Ok(Self::new(String::deserialize(d)?))
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn debug_masks_value() {
        let s = Secret::new("sk-ant-12345678".to_string());
        assert_eq!(format!("{s:?}"), "Secret([REDACTED])");
    }

    #[test]
    fn display_short_value_redacted() {
        let s = Secret::new("short".to_string());
        assert_eq!(format!("{s}"), "[REDACTED]");
    }

    #[test]
    fn display_long_value_partial() {
        let s = Secret::new("sk-ant-12345678".to_string());
        assert_eq!(format!("{s}"), "sk-a...5678");
    }

    #[test]
    fn expose_returns_inner() {
        let s = Secret::new("secret-key".to_string());
        assert_eq!(s.expose(), "secret-key");
    }
}
```

## 0.3 설정 검증

### oxicode-cli/src/settings_validation.rs (NEW)

```rust
//! 설정 검증 모듈.
//!
//! 애플리케이션 시작 시 모든 설정 값을 검증하여
//! 런타임 패닉을 사전에 방지한다.

use crate::settings::Settings;

/// 검증 결과
#[derive(Debug)]
pub struct ValidationReport {
    pub errors: Vec<ValidationError>,
    pub warnings: Vec<ValidationWarning>,
}

#[derive(Debug, Clone)]
pub struct ValidationError {
    pub field: String,
    pub message: String,
}

#[derive(Debug, Clone)]
pub struct ValidationWarning {
    pub field: String,
    pub message: String,
}

impl ValidationReport {
    pub fn is_valid(&self) -> bool {
        self.errors.is_empty()
    }
}

impl Settings {
    /// 현재 설정의 유효성을 검증한다.
    pub fn validate(&self) -> ValidationReport {
        let mut report = ValidationReport { errors: Vec::new(), warnings: Vec::new() };

        // — temperature ─────────────────────────────────
        if let Some(temp) = self.default_temperature {
            if !(0.0..=2.0).contains(&temp) {
                report.errors.push(ValidationError {
                    field: "default_temperature".into(),
                    message: format!("0.0~2.0 범위여야 함 (현재: {temp})"),
                });
            }
        }

        // — max_response_tokens ─────────────────────────
        if let Some(tokens) = self.max_response_tokens {
            if tokens == 0 {
                report.errors.push(ValidationError {
                    field: "max_response_tokens".into(),
                    message: "최소 1 이상이어야 함".into(),
                });
            } else if tokens > 128_000 {
                report.warnings.push(ValidationWarning {
                    field: "max_response_tokens".into(),
                    message: format!("매우 큰 값 ({tokens}). 대부분의 모델이 지원하지 않음"),
                });
            }
        }

        // — tool_timeout_seconds ────────────────────────
        if let Some(timeout) = self.tool_timeout_seconds {
            if timeout == 0 {
                report.errors.push(ValidationError {
                    field: "tool_timeout_seconds".into(),
                    message: "최소 1초 이상이어야 함".into(),
                });
            }
        }

        // — thinking_level ──────────────────────────────
        if let Some(ref level) = self.thinking_level {
            let valid = ["none", "minimal", "standard", "thorough"];
            if !valid.contains(&level.as_str()) {
                report.errors.push(ValidationError {
                    field: "thinking_level".into(),
                    message: format!(
                        "none, minimal, standard, thorough 중 하나여야 함 (현재: {level})"
                    ),
                });
            }
        }

        // — default_model 형식 ─────────────────────────
        if let Some(ref model) = self.default_model {
            if !model.contains('/') {
                report.warnings.push(ValidationWarning {
                    field: "default_model".into(),
                    message: format!("'provider/model' 형식 권장 (현재: {model})"),
                });
            }
        }

        report
    }
}
```

## 1.2 Agent → AgentLoop 통합 (1일)

### 현재 문제

중복: `stream_with_retry` 두 곳에 존재. `run()` 로직이 다르게 구현.
oxicode-cli는 `Agent`를 주로 사용. AgentLoop는 RPC/프로그래매틱 인터페이스.

### 설계: Agent = thin wrapper

```rust
// oxicode-agent/src/agent.rs (710줄 → ~130줄)

use crate::agent_loop::AgentLoop;
use crate::agent_loop::config::AgentLoopConfig;
use crate::shared::CompactionManager;

/// Agent는 AgentLoop의 mpsc 채널 기반 편의 래퍼.
///
/// 대부분의 로직은 내부 AgentLoop에 위임한다.
pub struct Agent {
    inner: AgentLoop,
}

impl Agent {
    pub fn new(provider: Arc<dyn Provider>, config: AgentConfig) -> Self {
        let state = SharedState::new();
        let tools = Arc::new(ToolRegistry::new());
        let loop_config = AgentLoopConfig::from(config);
        let inner = AgentLoop::new(provider, loop_config, tools, state, None);
        Self { inner }
    }

    /// mpsc 채널로 이벤트를 스트리밍하며 실행.
    pub async fn run_with_channel(
        &self,
        prompt: String,
        tx: mpsc::Sender<AgentEvent>,
    ) -> Result<Response> {
        let state = self.inner.state();
        state.update(|s| s.add_user_message(prompt.clone()));

        let result = self.inner.run(prompt, |event| {
            let _ = tx.blocking_send(event);
        }).await?;

        let response = self.build_response(&result);
        Ok(response)
    }

    /// 동기적 실행. 모든 이벤트를 모아서 반환.
    pub async fn run(&self, prompt: String) -> Result<(Response, Vec<AgentEvent>)> {
        let mut events = Vec::new();
        let (tx, mut rx) = mpsc::channel(100);

        let handle = tokio::spawn({
            let agent = self.clone(); // Agent를 Clone 가능하게
            async move { agent.run_with_channel(prompt, tx).await }
        });

        while let Some(event) = rx.recv().await {
            events.push(event);
        }
        let response = handle.await??;
        Ok((response, events))
    }

    // 위임 메서드들
    pub fn switch_model(&self, model_id: &str) -> Result<()> {
        self.inner.switch_model(model_id)
    }

    pub fn tools(&self) -> Arc<ToolRegistry> {
        self.inner.tools()
    }

    pub fn state(&self) -> AgentState {
        self.inner.state().get_state()
    }

    pub fn reset(&self) {
        self.inner.state().update(|s| s.reset());
    }

    pub fn add_tool<T: AgentTool + 'static>(&self, tool: T) {
        self.inner.tools().register(tool);
    }

    pub fn set_system_prompt(&self, prompt: String) {

//! 동적 라이브러리 로딩 전 검증.

use std::path::Path;
use sha2::{Sha256, Digest};

/// 확장 바이너리 검증 결과
pub struct ValidatedExtension {
    pub path: std::path::PathBuf,
    pub manifest: crate::types::ExtensionManifest,
    pub checksum: String,
}

/// 확장 로딩 전 검증 수행
pub fn validate_extension(path: &Path) -> Result<ValidatedExtension, crate::types::ExtensionError> {
    // 1. 파일 존재
    if !path.exists() {
        return Err(extension_error(path, "파일을 찾을 수 없음"));
    }

    // 2. 파일 크기 (0 또는 100MB 초과 차단)
    let len = std::fs::metadata(path)?.len();
    if len == 0 {
        return Err(extension_error(path, "빈 파일"));
    }
    if len > 100 * 1024 * 1024 {
        return Err(extension_error(path, "파일이 100MB 초과"));
    }

    // 3. 확장자 확인
    let ext = path.extension().and_then(|e| e.to_str()).unwrap_or("");
    let valid_ext = match std::env::consts::OS {
        "linux" => ext == "so",
        "macos" => ext == "dylib",
        "windows" => ext == "dll",
        _ => true,
    };
    if !valid_ext {
        return Err(extension_error(path, &format!("잘못된 확장자: .{ext}")));
    }

    // 4. 체크섬 계산
    let checksum = sha256_file(path)?;

    Ok(ValidatedExtension {
        path: path.to_path_buf(),
        manifest: Default::default(), // 실제로는 로딩 후 설정
        checksum,
    })
}

fn sha256_file(path: &Path) -> Result<String, std::io::Error> {
    let data = std::fs::read(path)?;
    let mut hasher = Sha256::new();
    hasher.update(&data);
    Ok(format!("{:x}", hasher.finalize()))
}

fn extension_error(path: &Path, reason: &str) -> crate::types::ExtensionError {
    crate::types::ExtensionError::LoadFailed {
        name: path.display().to_string(),
        reason: reason.to_string(),
    }
}

//! 파일 접근 경로 보안 검증.

use std::path::{Path, PathBuf};
use thiserror::Error;

#[derive(Debug, Error)]
pub enum PathSecurityError {
    #[error("경로를 찾을 수 없음: {0}")]
    NotFound(PathBuf),
    #[error("경로 순회 감지: {0}")]
    Traversal(PathBuf),
    #[error("작업 공간 밖 경로: {0}")]
    OutsideWorkspace(PathBuf),
}

/// 파일 접근 시 보안 검증 유틸.
pub struct PathGuard {
    root: PathBuf,
}

impl PathGuard {
    pub fn new(cwd: &Path) -> Self {
        let root = cwd.canonicalize().unwrap_or_else(|_| cwd.to_path_buf());
        Self { root }
    }

    /// 경로가 작업 공간 내에 있는지 확인.
    pub fn validate(&self, path: &Path) -> Result<PathBuf, PathSecurityError> {
        // 1. 순회 방지
        if path.components().any(|c| c.as_os_str() == "..") {
            return Err(PathSecurityError::Traversal(path.to_path_buf()));
        }

        // 2. canonicalize로 실제 경로 확인 (심볼릭 링크 해소)
        let canonical = path.canonicalize()
            .map_err(|_| PathSecurityError::NotFound(path.to_path_buf()))?;

        // 3. 루트 내부인지 확인
        if !canonical.starts_with(&self.root) {
            return Err(PathSecurityError::OutsideWorkspace(canonical));
        }

        Ok(canonical)
    }
}

```rust
//! Graceful shutdown 조정자.
//!
//! Ctrl+C를 2단계로 처리:
//! 1차: 진행 중인 작업 완료 후 종료
//! 2차: 즉시 종료

use std::sync::atomic::{AtomicU8, Ordering};
use tokio::sync::broadcast;

/// 0=Running, 1=Draining, 2=Forced
const RUNNING: u8 = 0;
const DRAINING: u8 = 1;
const FORCED: u8 = 2;

#[derive(Debug, Clone)]
pub enum ShutdownSignal {
    Graceful,
    Force,
}

pub struct ShutdownCoordinator {
    tx: broadcast::Sender<ShutdownSignal>,
    state: std::sync::Arc<AtomicU8>,
}

impl ShutdownCoordinator {
    pub fn new() -> Self {
        let (tx, _) = broadcast::channel(2);
        Self { tx, state: std::sync::Arc::new(AtomicU8::new(RUNNING)) }
    }

    /// SIGINT/SIGTERM 리스너 시작
    pub fn listen(&self) {
        let tx = self.tx.clone();
        let state = self.state.clone();

        tokio::spawn(async move {
            // 첫 번째 Ctrl+C
            tokio::signal::ctrl_c().await.ok();
            if state.compare_exchange(RUNNING, DRAINING, Ordering::SeqCst, Ordering::SeqCst).is_ok() {
                tracing::info!("Graceful shutdown 요청 (다시 Ctrl+C로 강제 종료)");
                let _ = tx.send(ShutdownSignal::Graceful);
            }

            // 두 번째 Ctrl+C
            tokio::signal::ctrl_c().await.ok();
            state.store(FORCED, Ordering::SeqCst);
            tracing::warn!("강제 종료 요청");
            let _ = tx.send(ShutdownSignal::Force);
        });
    }

    pub fn subscribe(&self) -> broadcast::Receiver<ShutdownSignal> {
        self.tx.subscribe()
    }

    pub fn is_draining(&self) -> bool {
        self.state.load(Ordering::SeqCst) >= DRAINING
    }

    pub fn is_forced(&self) -> bool {
        self.state.load(Ordering::SeqCst) == FORCED
    }
}
```

```rust
/// 모든 확장이 반드시 구현하는 기본 트레이트.
pub trait Extension: Send + Sync {
    fn name(&self) -> &str;
    fn description(&self) -> &str;
    fn manifest(&self) -> ExtensionManifest {
        ExtensionManifest::new(self.name(), "0.0.0")
            .with_description(self.description())
    }

    /// 로딩/언로딩 라이프사이클.
    fn on_load(&self, _ctx: &ExtensionContext) {}
    fn on_unload(&self) {}
}

/// 도구 제공 트레이트. 필요한 확장만 구현.
pub trait ToolProvider: Extension {
    fn register_tools(&self) -> Vec<Arc<dyn AgentTool>> { vec![] }
}

/// 세션 이벤트 훅. 필요한 확장만 구현.
pub trait SessionHooks: Extension {
    fn on_session_start(&self, _id: &str) {}
    fn on_session_end(&self, _id: &str) {}
}

/// 스트리밍 이벤트 훅.
pub trait StreamHooks: Extension {
    fn on_text_delta(&self, _delta: &str) {}
    fn on_thinking_delta(&self, _delta: &str) {}
}

/// 에러/복구 훅.
pub trait ErrorHooks: Extension {
    fn on_error(&self, _error: &anyhow::Error) -> Result<()> { Ok(()) }
}
```

### 레지스트리 수정

```rust
// oxicode-cli/src/extensions/registry.rs
impl ExtensionRegistry {
    pub fn load(&mut self, path: &Path) -> Result<ExtensionId, ExtensionError> {
        let ext = load_library(path)?;

        // 모든 Extension은 name() 보유
        let id = ExtensionId::new(ext.name());

        // ToolProvider이면 도구 등록
        if let Some(tp) = ext.as_any().downcast_ref::<dyn ToolProvider>() {
            for tool in tp.register_tools() {
                self.tools.register(tool);
            }
        }

        // SessionHooks이면 리스너 등록
        if let Some(sh) = ext.as_any().downcast_ref::<dyn SessionHooks>() {
            // 이벤트 구독
        }

        self.extensions.insert(id, ext);
        Ok(id)
    }
}
```
